final class _Message extends StatelessWidget {
  const _Message(this.text);
  final String text;
  @override Widget build(BuildContext context) => Padding(padding: const EdgeInsets.all(36), child: Center(child: Text(text, textAlign: TextAlign.center)));
}

  ExternalSourceKind.gelbooru => 'Gelbooru',
  ExternalSourceKind.pixiv => 'Pixiv',
};
